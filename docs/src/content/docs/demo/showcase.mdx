                <div style={{ minWidth: 0, flex: 1, margin: 0 }}>

                </div>

              </header>

                <span>♥ {tweet.favorite_count.toLocaleString()}</span>

              </footer>
            </div>
          </article>
        ) : (
          <div role="status" style={{ padding: '20px', borderRadius: '20px', background: '#fff1f2', color: '#9f1239' }}>Post unavailable</div>
        )}
